from pathlib import Path
import argparse, datetime, hashlib, json, subprocess

parser = argparse.ArgumentParser()
parser.add_argument("--snapshot", action="store_true", help="record the pre-task source hash baseline")
args = parser.parse_args()

run = Path(__file__).parent
source = Path("/Users/davidarendash/Documents/Projects/Unity/HeadsetHolodeckDev")
project = run / "project"
evidence = source / "docs/stabilization/evidence-04"
evidence.mkdir(parents=True, exist_ok=True)

SKIP_NAMES = {".env", "worldlabs_env.txt"}
ADAPTERS = {
    "Assets/App/Editor/StabilizationBaseline01.cs",
    "Assets/App/Editor/StabilizationSaveRisk03.cs",
    "Assets/App/Editor/StabilizationSaveRisk04.cs",
    "Assets/App/Editor/StabilizationTransitions02.cs",
}


def digest(path):
    h = hashlib.sha256()
    with open(path, "rb") as handle:
        for chunk in iter(lambda: handle.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def project_files(root, prefixes):
    names = subprocess.check_output(
        ["git", "ls-files", "-z", "--cached", "--others", "--exclude-standard"], cwd=source
    ).decode().split("\0")
    for name in names:
        if not name or not name.startswith(prefixes):
            continue
        if Path(name).name in SKIP_NAMES:
            continue
        candidate = root / name
        if candidate.is_file():
            yield name, candidate


prefixes = ("Assets/", "Packages/", "ProjectSettings/")
source_hashes = {name: digest(path) for name, path in project_files(source, prefixes)}

if args.snapshot:
    target = evidence / "source-hashes-before.json"
    if target.exists():
        raise SystemExit("Baseline snapshot already exists; refusing to overwrite")
    target.write_text(json.dumps(source_hashes, indent=1, sort_keys=True) + "\n")
    print(json.dumps({"snapshot": str(target), "files": len(source_hashes)}, indent=2))
    raise SystemExit(0)

copy_hashes = {name: digest(path) for name, path in project_files(project, prefixes)}

only_in_source = sorted(n for n in source_hashes if n not in copy_hashes and n not in ADAPTERS)
only_in_copy = sorted(n for n in copy_hashes if n not in source_hashes and n not in ADAPTERS)
mismatched = sorted(n for n in source_hashes if n in copy_hashes and source_hashes[n] != copy_hashes[n])

comparison = {
    "time": datetime.datetime.now().astimezone().isoformat(),
    "compared_prefixes": list(prefixes),
    "source_files": len(source_hashes),
    "copy_files": len(copy_hashes),
    "installed_adapters_in_copy": sorted(ADAPTERS),
    "only_in_source": only_in_source,
    "only_in_copy": only_in_copy,
    "mismatched": mismatched,
    "match": not only_in_source and not only_in_copy and not mismatched,
}
(evidence / "final-source-comparison.json").write_text(json.dumps(comparison, indent=2) + "\n")
(evidence / "verification-source-hashes.json").write_text(json.dumps(source_hashes, indent=1, sort_keys=True) + "\n")

baseline = json.loads((evidence / "source-hashes-before.json").read_text())
current = {}
for name in baseline:
    path = source / name
    current[name] = digest(path) if path.is_file() else None

changed = sorted(n for n in baseline if current[n] is not None and current[n] != baseline[n])
removed = sorted(n for n in baseline if current[n] is None)

now = subprocess.check_output(
    ["git", "ls-files", "-z", "--cached", "--others", "--exclude-standard"], cwd=source
).decode().split("\0")
added = sorted(
    n for n in now
    if n and n not in baseline and n.startswith(prefixes)
    and Path(n).name not in SKIP_NAMES and (source / n).is_file()
)

preservation = {
    "time": datetime.datetime.now().astimezone().isoformat(),
    "baseline_snapshot": "source-hashes-before.json",
    "baseline_files": len(baseline),
    "changed": changed,
    "removed": removed,
    "added": added,
    "untouched": len(baseline) - len(changed) - len(removed),
}
(evidence / "preservation-final.json").write_text(json.dumps(preservation, indent=2) + "\n")
print(json.dumps({"comparison": comparison["match"], "preservation": preservation}, indent=2))
