using System;
using System.IO;
using System.Reflection;
using UnityEditor;
using UnityEditor.Build.Reporting;
using UnityEditor.SceneManagement;
using UnityEngine;
using WorldLabs.Editor;

/// <summary>
/// Work order 04 end-to-end fixture. Drives the real BuildEnvCopier build callbacks through
/// reflection in an ISOLATED copy of the project, with a throwaway .env of fake values written into
/// that copy's project root. The user's real checkout and real .env are never opened: the harness
/// refuses to run unless the project path is an isolated temp copy, and it removes the fake .env and
/// the embedded copy before it exits.
/// </summary>
public static class StabilizationCredentials04
{
    const string FakeOpenAi = "fake-openai-value-not-a-real-key";
    const string FakeWorldLabs = "fake-worldlabs-value-not-a-real-key";
    const string FakeUnrelated = "fake-personal-laptop-value";
    const string FakeBaseUrl = "https://fixture.example/base";

    static string _resultPath;
    static int _failures;

    static string Arg(string name)
    {
        string[] args = Environment.GetCommandLineArgs();
        for (int i = 0; i < args.Length - 1; i++) if (args[i] == name) return args[i + 1];
        throw new ArgumentException("Missing " + name);
    }

    static void Record(string message)
    {
        File.AppendAllText(_resultPath, DateTime.UtcNow.ToString("o") + " " + message + "\n");
        Debug.Log("[Credentials04] " + message);
    }

    static void Check(string label, bool condition, string detail = "")
    {
        if (!condition) _failures++;
        Record((condition ? "PASS " : "FAIL ") + label + (string.IsNullOrEmpty(detail) ? "" : " :: " + detail));
    }

    public static void Run()
    {
        if (!Application.isBatchMode)
            throw new InvalidOperationException("Only run in an isolated batch editor.");

        _resultPath = Arg("-credentialsResult");
        EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
        int code = 0;
        try
        {
            Record("ENV unity=" + Application.unityVersion + " platform=" + Application.platform);

            string projectRoot = Directory.GetParent(Application.dataPath).FullName;
            Record("PROJECT-ROOT " + projectRoot);
            if (projectRoot == null || projectRoot.IndexOf("/private/tmp/holodeck-", StringComparison.Ordinal) < 0)
            {
                Record("FAIL refusing to run outside an isolated temp copy");
                EditorApplication.Exit(2);
                return;
            }

            string envPath = Path.Combine(projectRoot, ".env");
            string embedded = Path.Combine("Assets", "Resources", "WorldLabs", "worldlabs_env.txt");
            string embeddedFull = Path.GetFullPath(embedded);

            if (File.Exists(envPath))
            {
                Record("FAIL isolated copy already has a .env; refusing to overwrite it");
                EditorApplication.Exit(2);
                return;
            }

            Type copier = FindCopierType();
            Check("copier-type-found", copier != null);
            if (copier == null)
            {
                EditorApplication.Exit(1);
                return;
            }

            BuildCredentialSettings.Clear();
            RunScenarioDistribution(copier, envPath, embedded, embeddedFull);
            RunScenarioUndeclared(copier, envPath, embedded, embeddedFull);
            RunScenarioPrivateThenCleanup(copier, envPath, embedded, embeddedFull);
            RunScenarioStalePrivateCopy(copier, envPath, embedded, embeddedFull);

            Record(_failures == 0 ? "CREDENTIALS COMPLETE" : "CREDENTIALS COMPLETE WITH FAILURES");
        }
        catch (Exception error)
        {
            code = 1;
            Record("FAIL " + error.GetBaseException());
        }
        finally
        {
            try { BuildCredentialSettings.Clear(); } catch (Exception) { }
            try { Environment.SetEnvironmentVariable(BuildEnvCredentials.ModeEnvironmentVariable, null); } catch (Exception) { }
        }
        EditorApplication.Exit(_failures == 0 ? code : 1);
    }

    static Type FindCopierType()
    {
        foreach (Assembly assembly in AppDomain.CurrentDomain.GetAssemblies())
        {
            Type type = assembly.GetType("WorldLabs.Editor.BuildEnvCopier");
            if (type != null) return type;
        }
        return null;
    }

    static object NewCopier(Type copier) => Activator.CreateInstance(copier, true);

    static Exception InvokePreprocess(Type copier, object instance, string mode)
    {
        Environment.SetEnvironmentVariable(BuildEnvCredentials.ModeEnvironmentVariable, mode);
        MethodInfo preprocess = copier.GetMethod("OnPreprocessBuild", BindingFlags.Public | BindingFlags.Instance);
        try
        {
            preprocess.Invoke(instance, new object[] { null });
            return null;
        }
        catch (TargetInvocationException wrapper)
        {
            return wrapper.InnerException ?? wrapper;
        }
    }

    static void InvokePostprocess(Type copier, object instance)
    {
        MethodInfo postprocess = copier.GetMethod("OnPostprocessBuild", BindingFlags.Public | BindingFlags.Instance);
        postprocess.Invoke(instance, new object[] { BuildTarget.Android, "fixture.apk" });
    }

    static void WriteFixtureEnv(string envPath)
    {
        File.WriteAllText(envPath,
            "# fixture .env — fake values only\n" +
            "OPENAI_API_KEY=" + FakeOpenAi + "\n" +
            "WORLDLABS_API_KEY=" + FakeWorldLabs + "\n" +
            "MY_PERSONAL_LAPTOP=" + FakeUnrelated + "\n" +
            "THREEDAISTUDIO_BASE_URL=" + FakeBaseUrl + "\n");
    }

    static void DeleteQuietly(string path)
    {
        try { if (File.Exists(path)) File.Delete(path); } catch (Exception) { }
        try { if (File.Exists(path + ".meta")) File.Delete(path + ".meta"); } catch (Exception) { }
    }

    static void RunScenarioDistribution(Type copier, string envPath, string embedded, string embeddedFull)
    {
        WriteFixtureEnv(envPath);
        Exception error = InvokePreprocess(copier, NewCopier(copier), "distribution");
        Check("distribution-preprocess-succeeds", error == null, error?.GetType().Name);
        Check("distribution-embedded-file-written", File.Exists(embeddedFull));

        string content = File.Exists(embeddedFull) ? File.ReadAllText(embeddedFull) : string.Empty;
        Check("distribution-keeps-safe-configuration", content.Contains(FakeBaseUrl));
        Check("distribution-drops-credentials", !content.Contains(FakeOpenAi) && !content.Contains(FakeWorldLabs));
        Check("distribution-drops-unrelated-entries", !content.Contains(FakeUnrelated));
        DeleteQuietly(embeddedFull);
        DeleteQuietly(envPath);
    }

    static void RunScenarioUndeclared(Type copier, string envPath, string embedded, string embeddedFull)
    {
        WriteFixtureEnv(envPath);
        Exception error = InvokePreprocess(copier, NewCopier(copier), null);
        Check("undeclared-build-stops", error is BuildCredentialModeException, error?.GetType().Name);
        Check("undeclared-build-says-how-to-choose",
            error != null && error.Message.Contains("Build Credentials") && error.Message.Contains("distribution"));
        Check("undeclared-build-embeds-nothing", !File.Exists(embeddedFull));
        DeleteQuietly(embeddedFull);
        DeleteQuietly(envPath);
    }

    static void RunScenarioPrivateThenCleanup(Type copier, string envPath, string embedded, string embeddedFull)
    {
        WriteFixtureEnv(envPath);
        object instance = NewCopier(copier);
        Exception error = InvokePreprocess(copier, instance, "private");
        Check("private-preprocess-succeeds", error == null, error?.GetType().Name);
        string content = File.Exists(embeddedFull) ? File.ReadAllText(embeddedFull) : string.Empty;
        Check("private-embeds-credentials", content.Contains(FakeOpenAi) && content.Contains(FakeWorldLabs));
        Check("private-leaves-no-temporary-file",
            Directory.GetFiles(Path.GetDirectoryName(embeddedFull), "*.tmp", SearchOption.TopDirectoryOnly).Length == 0);

        InvokePostprocess(copier, instance);
        Check("postprocess-removes-embedded-file", !File.Exists(embeddedFull));
        Check("postprocess-removes-embedded-meta", !File.Exists(embeddedFull + ".meta"));
        DeleteQuietly(embeddedFull);
        DeleteQuietly(envPath);
    }

    static void RunScenarioStalePrivateCopy(Type copier, string envPath, string embedded, string embeddedFull)
    {
        WriteFixtureEnv(envPath);
        Directory.CreateDirectory(Path.GetDirectoryName(embeddedFull));
        File.WriteAllText(embeddedFull, "# left by a crashed private build\nOPENAI_API_KEY=" + FakeOpenAi + "\n");

        Exception error = InvokePreprocess(copier, NewCopier(copier), "distribution");
        Check("stale-copy-preprocess-succeeds", error == null, error?.GetType().Name);
        string content = File.Exists(embeddedFull) ? File.ReadAllText(embeddedFull) : string.Empty;
        Check("stale-copy-credentials-do-not-survive", !content.Contains(FakeOpenAi));

        InvokePostprocess(copier, NewCopier(copier));
        Check("cleanup-removes-everything", !File.Exists(embeddedFull));
        DeleteQuietly(embeddedFull);
        DeleteQuietly(envPath);

        string resourcesDir = Path.GetDirectoryName(embeddedFull);
        Check("cleanup-left-no-stray-resources-dir", !Directory.Exists(resourcesDir));
    }
}
