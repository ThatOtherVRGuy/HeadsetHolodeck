public static class StabilizationBaseline01 {
 static string Arg(string name,string fallback) { var a=Environment.GetCommandLineArgs(); for(int i=0;i<a.Length-1;i++)if(a[i]==name)return a[i+1]; return fallback; }
 static void Record(string message) { File.AppendAllText(Arg("-baselineResult","baseline-results.txt"),DateTime.UtcNow.ToString("o")+" "+message+"\n"); Debug.Log("[Baseline01] "+message); }
 public static void Validate() {
  Record("STATE unity="+Application.unityVersion+" batch="+Application.isBatchMode+" playing="+EditorApplication.isPlaying+" compiling="+EditorApplication.isCompiling+" updating="+EditorApplication.isUpdating+" target="+EditorUserBuildSettings.activeBuildTarget);
  var window=ScriptableObject.CreateInstance<HeadsetHolodeck.Editor.HeadsetHolodeckInstallValidator>();
  try { var t=window.GetType(); t.GetMethod("Refresh",BindingFlags.Instance|BindingFlags.NonPublic).Invoke(window,null); Record("VALIDATOR "+t.GetField("_summary",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(window)); }
  finally { UnityEngine.Object.DestroyImmediate(window); }
 }
